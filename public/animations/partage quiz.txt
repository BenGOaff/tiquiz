<!-- Scroll reveal + URL typing compatible Systeme.io -->
<script>
(function(){
  var done=false;
  var typed=false;
  var url="https://app.tiquiz.com/sandra-costa/formation-canine";

  function startTyping(){
    if(typed)return;
    typed=true;
    var el=document.getElementById('tqz-sh-url');
    if(!el)return;
    var i=0;
    var iv=setInterval(function(){
      if(i<url.length){
        el.textContent+=url.charAt(i);
        i++;
      }else{
        clearInterval(iv);
      }
    },22);
  }

  function check(){
    var el=document.getElementById('tqz-sh-widget');
    if(!el||done)return;
    var rect=el.getBoundingClientRect();
    var winH=window.innerHeight||document.documentElement.clientHeight;
    if(rect.top<winH*0.85){
      el.classList.add('tqz-visible');
      done=true;
      window.removeEventListener('scroll',check);
      window.removeEventListener('resize',check);
      setTimeout(startTyping,800);
    }
  }
  window.addEventListener('scroll',check,{passive:true});
  window.addEventListener('resize',check,{passive:true});
  var polls=0;
  var poller=setInterval(function(){
    check();
    polls++;
    if(done||polls>33)clearInterval(poller);
  },300);
  check();
})();
</script>
